# Table

A component which allows you to render a table, each row can have custom elements defined to be used for actions. The table can also be sorted in asc / desc order and filter results with as search input  
Table is UI/UX agnostic so you need to provide your style to have the look and feel you prefer.  
When you import the table component without providing any className or style associated will looks as following:

<Canvas of={TableStories.Unstyled} />

An example with all the available properties is:

```js
const ELEMENT_DATA: PeriodicElement[] = [
  {
    id: 1,
    position: 3,
    name: 'Hydrogen',
    weight: 1.0079,
    symbol: 'H',
    actions: (
      <>
        <button
          className="btn btn-primary"
          name="add"
          style={{ marginRight: '5px' }}
        >
          Add
        </button>
        <button className="btn btn-danger" name="delete">
          Delete
        </button>
      </>
    ),
  },
];

const handleSelect = row => {
  alert(row);
};
const handleCellClick = (target: any, value: any) => {
  if (target.name === 'delete') {
    alert(value);
  } else if (target.name === 'edit') {
    alert(value);
  }
};

<Table
  dataSource={ELEMENT_DATA}
  customHeaderLabels={['Test', 'position', 'name', 'weight', 'symbol', 'actions']}
  columnsToOmit={['symbol']}
  onSelect={handleSelect}
  handleCellClick={handleCellClick}
  tableClassName="table"
  theadClassName="thead"
  trClassName="header-row"
  thClassName="th"
  tbodyClassName="tbody"
  tdClassName="td"
  selectedRowClassName="trSelected"
  withOrderBy={true}
  withSearch={true}
  searchProps={{
    placeholder: 'Search...'
    className: 'searchClass'
  }}
  sortAscIcon={
    <FontAwesomeIcon icon={faArrowUp} style={{ marginLeft: '4px' }} />
  }
  sortDescIcon={
    <FontAwesomeIcon icon={faArrowDown} style={{ marginLeft: '4px' }} />
  }
  columnsWidth={['100px']}
  tabIndex={0}
  paginator={{
    currentPageClassName:"current-page",
    nextButton:"Next",
    nextButtonClassName:"next-button",
    pageNumbersClassName:"buttons",
    paginatorClassName:"paginator",
    previousButtonClassName:"previous-button",
    previousButton:"Prev",
    totalPages:14,
    currentPage: 2,
    siblingCount: 1, 
    boundaryCount: 1, 
    onPageChange: (page: number) => {},
  }}
```

 **Note** that there is the option to add a Paginator at the bottom of the table. This uses the Paginator component which is also part of this library. Documentation for the Paginator component can be found here: <a href="/?path=/docs/dcxlibrary-layout-paginator-documentation--docs">Paginator Component Docs</a>

If you want to connect the table with the Paginator component you can use the `onPageChange` prop to handle the page change and pass it to the Paginator component. The Paginator component will then handle the page change and update the table accordingly.

```jsx
const [currentPage, setCurrentPage] = useState(1);
const rowPerPage = 2;
const pageChange = page => {
  setCurrentPage(page);
};
const finalRow = currentPage * rowPerPage;
const firstRow = finalRow - rowPerPage;
const visibleRows = args.dataSource.slice(firstRow, finalRow);
  
return (
  <Table
    //all the table props
    paginator= {{
      totalPages: 2,
      currentPage: rowPerPage,
      currentPageClassName: 'current-page',
      paginatorClassName: 'paginator',
      previousButton: '<',
      previousButtonClassName: 'previous-button',
      nextButton: '>',
      nextButtonClassName: 'next-button',
      pageNumbersClassName: 'buttons',
      siblingCount: 1,
      boundaryCount: 1,
    }}
    onPageChange={pageChange}
  />
```
